// Форматирование числа с пробелами (только визуальное)
function formatNumberInput(input) {
  input.addEventListener('input', () => {
    const raw     = input.value.replace(/\s/g, '');
    const numeric = raw.replace(/[^\d]/g, '');
    if (!numeric) { input.value = ''; return; }
    input.value = parseInt(numeric, 10).toLocaleString('ru-RU');
  });
}

formatNumberInput(container.querySelector('#inputAmount'));

// Получить чистое число из поля с пробелами
function parseFormattedNumber(val) {
  return parseFloat(val.replace(/\s/g, '').replace(',', '.'));
}

function getDaysInMonth(year, month) {
  return new Date(year, month + 1, 0).getDate();
}

function getDaysInYear(year) {
  return ((year % 4 === 0 && year % 100 !== 0) || year % 400 === 0) ? 366 : 365;
}

function addMonths(date, months) {
  const d = new Date(date);
  d.setMonth(d.getMonth() + months);
  return d;
}

function formatDate(date) {
  return date.toLocaleDateString('ru-RU', { day: '2-digit', month: '2-digit', year: 'numeric' });
}

function formatMoney(amount) {
  return amount.toLocaleString('ru-RU', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' ₽';
}

  if (isNaN(rate)   || rate   <= 0)   { errorEl.textContent = '❌ Укажи процентную ставку'; return; }
  if (!dateStr)                        { errorEl.textContent = '❌ Укажи дату открытия'; return; }
  if (doReplenish && (isNaN(replenish) || replenish <= 0)) { errorEl.textContent = '❌ Укажи сумму пополнения'; return; }

  const startDate  = new Date(dateStr);
  const annualRate = rate / 100;

  let currentAmount    = amount;
  let totalInterest    = 0;
  let totalReplenished = 0;
  const rows           = [];

  for (let i = 0; i < months; i++) {
    const periodStart = addMonths(startDate, i);
    const periodEnd   = addMonths(startDate, i + 1);

    const daysInYear  = getDaysInYear(periodStart.getFullYear());

    const interest = currentAmount * annualRate * (daysInMonth / daysInYear);
    totalInterest += interest;

    // Баланс после начисления процентов
    if (capitalize) {
      currentAmount += interest;
    }

    // Баланс после пополнения (кроме последнего месяца)
