function attachFormatter(input) {
  input.addEventListener('input', () => {
    const numeric = input.value.replace(/\s/g, '').replace(/[^\d]/g, '');
    if (!numeric) { input.value = ''; return; }
    input.value = parseInt(numeric, 10).toLocaleString('ru-RU');
  });
}
['#inputPrice','#inputDown','#inputOneTime','#inputEarlyAmount'].forEach(id => {
  attachFormatter(container.querySelector(id));
});

function parseNum(val) {
  return parseFloat((val || '').replace(/\s/g, '').replace(',', '.')) || 0;
}

// Процент взноса
function updateDownPercent() {
  const price = parseNum(container.querySelector('#inputPrice').value);
  const down  = parseNum(container.querySelector('#inputDown').value);
  const el    = container.querySelector('#downPercent');
  if (price > 0 && down > 0) {
    el.textContent = `${((down / price) * 100).toFixed(1)}% от стоимости`;
  } else {
    el.textContent = '';
  }
}

function formatMoney(n) {
  return n.toLocaleString('ru-RU', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) + ' ₽';
}
function formatMoneyShort(n) {
  return n.toLocaleString('ru-RU', { minimumFractionDigits: 0, maximumFractionDigits: 0 }) + ' ₽';
}
function addMonths(date, m) {
  const d = new Date(date);
  d.setMonth(d.getMonth() + m);
  return d;
}
function formatDate(date) {
  return date.toLocaleDateString('ru-RU', { day: '2-digit', month: '2-digit', year: 'numeric' });
}

container.querySelector('#btnCalc').addEventListener('click', () => {
  const errorEl  = container.querySelector('#error');
  const resultEl = container.querySelector('#result');
  errorEl.textContent = '';
  resultEl.innerHTML  = '';

  const price         = parseNum(container.querySelector('#inputPrice').value);
  const down          = parseNum(container.querySelector('#inputDown').value);
  const rate          = parseFloat(container.querySelector('#inputRate').value);
  const years         = parseInt(container.querySelector('#inputYears').value);
  const dateStr       = container.querySelector('#inputDate').value;
  const payType       = container.querySelector('#selectType').value;
  const insProp       = parseFloat(container.querySelector('#inputInsuranceProp').value) || 0;
  const insLife       = parseFloat(container.querySelector('#inputInsuranceLife').value) || 0;
  const oneTime       = parseNum(container.querySelector('#inputOneTime').value);
  const doEarly       = container.querySelector('#checkEarly').checked;
  const earlyAmount   = parseNum(container.querySelector('#inputEarlyAmount').value);
  const earlyFrom     = parseInt(container.querySelector('#inputEarlyFrom').value) || 1;
  const earlyFreq     = container.querySelector('#selectEarlyFreq').value;
  const earlyGoal     = container.querySelector('#selectEarlyGoal').value;

  // Валидация
  if (!price || price <= 0)  { errorEl.textContent = '❌ Укажи стоимость недвижимости'; return; }
  if (down >= price)          { errorEl.textContent = '❌ Взнос не может быть больше стоимости'; return; }
  if (!rate || rate <= 0)     { errorEl.textContent = '❌ Укажи процентную ставку'; return; }
  if (!years || years <= 0)   { errorEl.textContent = '❌ Укажи срок кредита'; return; }
  if (!dateStr)               { errorEl.textContent = '❌ Укажи дату первого платежа'; return; }
  if (doEarly && (!earlyAmount || earlyAmount <= 0)) { errorEl.textContent = '❌ Укажи сумму досрочного платежа'; return; }

  const principal   = price - down;
  const monthRate   = rate / 100 / 12;
  const totalMonths = years * 12;
  const startDate   = new Date(dateStr);

  // Аннуитетный платёж (базовый без досрочки)
  const annuityPayment = monthRate === 0
    ? principal / totalMonths
    : principal * monthRate * Math.pow(1 + monthRate, totalMonths)
      / (Math.pow(1 + monthRate, totalMonths) - 1);

  // Функция: применять ли досрочку в этом месяце
  function applyEarly(monthNum) {
    if (!doEarly || monthNum < earlyFrom) return 0;
    const offset = monthNum - earlyFrom;
    if (earlyFreq === 'once'        && offset === 0) return earlyAmount;
    if (earlyFreq === 'monthly')                      return earlyAmount;
    if (earlyFreq === 'quarterly'   && offset % 3 === 0) return earlyAmount;
    if (earlyFreq === 'yearly'      && offset % 12 === 0) return earlyAmount;
    return 0;
  }

  // Расчёт графика
  let balance      = principal;
  let monthPayment = annuityPayment;
  const rows       = [];
  let totalPaid    = 0;
  let totalInterestPaid = 0;
  let totalEarlyPaid    = 0;
  let totalInsPaid      = 0;

  for (let i = 1; i <= totalMonths; i++) {
    if (balance <= 0.01) break;

    const date          = addMonths(startDate, i - 1);
    const interestPart  = balance * monthRate;

    let principalPart, payment;

    if (payType === 'annuity') {
      // Пересчитываем аннуитет если была досрочка с целью уменьшить платёж
      payment       = Math.min(monthPayment, balance + interestPart);
      principalPart = payment - interestPart;
    } else {
      // Дифференцированный
      const basePrincipal = principal / totalMonths;
      principalPart       = Math.min(basePrincipal, balance);
      payment             = principalPart + interestPart;
    }

    // Страховка (от остатка долга, раз в год — делим на 12)
    const insMonth = balance * (insProp + insLife) / 100 / 12;
    totalInsPaid  += insMonth;

    balance -= principalPart;

    // Досрочное погашение
    const earlyThis = applyEarly(i);
    let actualEarly = 0;
    if (earlyThis > 0 && balance > 0) {
      actualEarly    = Math.min(earlyThis, balance);
      balance       -= actualEarly;
      totalEarlyPaid += actualEarly;

      // Пересчёт платежа если цель — уменьшить платёж
      if (earlyGoal === 'payment' && payType === 'annuity' && balance > 0) {
        const remainingMonths = totalMonths - i;
        if (remainingMonths > 0) {
          monthPayment = balance * monthRate * Math.pow(1 + monthRate, remainingMonths)
            / (Math.pow(1 + monthRate, remainingMonths) - 1);
        }
      }
    }

    totalInterestPaid += interestPart;
    totalPaid         += payment + actualEarly + insMonth;

    rows.push({
      num:          i,
      date:         formatDate(date),
      payment:      payment,
      principal:    principalPart,
      interest:     interestPart,
      early:        actualEarly,
      insurance:    insMonth,
      balance:      Math.max(balance, 0),
    });

    if (balance <= 0.01) break;
  }

  const actualMonths  = rows.length;
  const actualYears   = Math.floor(actualMonths / 12);
  const actualMo      = actualMonths % 12;
  const savedMonths   = totalMonths - actualMonths;
  const overPayment   = totalInterestPaid + totalInsPaid + oneTime;
  const totalCost     = principal + overPayment;

        <span style="font-size:15px;font-weight:700;color:#cdd6f4;word-break:break-word;">${formatMoneyShort(principal)}</span>
      </div>
      <div style="background:#313244;border-radius:6px;padding:12px;display:flex;flex-direction:column;gap:4px;">
        <span style="font-size:12px;color:#a6adc8;">${payType === 'annuity' ? 'Ежемесячный платёж' : 'Первый платёж'}</span>
        <span style="font-size:15px;font-weight:700;color:#89b4fa;word-break:break-word;">${formatMoneyShort(rows[0].payment)}</span>
        ${payType === 'differentiated' ? `<span style="font-size:11px;color:#6c7086;">последний: ${formatMoneyShort(rows[rows.length-1].payment)}</span>` : ''}
      </div>
      <div style="background:#313244;border-radius:6px;padding:12px;display:flex;flex-direction:column;gap:4px;">
        <span style="font-size:12px;color:#a6adc8;">Переплата по %</span>
        <span style="font-size:15px;font-weight:700;color:#f38ba8;word-break:break-word;">+${formatMoneyShort(totalInterestPaid)}</span>
      </div>
      <div style="background:#313244;border-radius:6px;padding:12px;display:flex;flex-direction:column;gap:4px;">
        <span style="font-size:12px;color:#a6adc8;">Итого выплат</span>
        <span style="font-size:15px;font-weight:700;color:#cba6f7;word-break:break-word;">${formatMoneyShort(totalCost)}</span>
      </div>
    </div>
  `;

  // Блок деталей

      <span style="color:#a6adc8;">Досрочно погашено:</span>
      <span style="color:#a6e3a1;font-weight:600;">${formatMoney(totalEarlyPaid)}</span>
    </div>
    <div style="display:flex;justify-content:space-between;flex-wrap:wrap;gap:4px;">
      <span style="color:#a6adc8;">Сэкономлено на %:</span>
      <span style="color:#a6e3a1;font-weight:600;">~${formatMoney((annuityPayment * totalMonths - annuityPayment * totalMonths) + (totalMonths - actualMonths) * rows[Math.min(actualMonths, totalMonths)-1]?.interest || 0)}</span>
    </div>
    <div style="display:flex;justify-content:space-between;flex-wrap:wrap;gap:4px;">
      <span style="color:#a6adc8;">Срок сокращён на:</span>
      <span style="color:#a6e3a1;font-weight:600;">${savedMonths} мес.</span>
    </div>
  ` : '';

  const insRow = (insProp > 0 || insLife > 0) ? `
    <div style="display:flex;justify-content:space-between;flex-wrap:wrap;gap:4px;">
      <span style="color:#a6adc8;">Страховка за период:</span>
      <span style="color:#fab387;font-weight:600;">${formatMoney(totalInsPaid)}</span>
    </div>
  ` : '';

  const oneTimeRow = oneTime > 0 ? `
    <div style="display:flex;justify-content:space-between;flex-wrap:wrap;gap:4px;">
      <span style="color:#a6adc8;">Единовременные расходы:</span>

      <div style="display:flex;justify-content:space-between;flex-wrap:wrap;gap:4px;">
        <span style="color:#a6adc8;">Фактический срок:</span>

        <span style="color:#f38ba8;font-weight:700;">${formatMoney(principal + totalInterestPaid + totalInsPaid + oneTime)}</span>
      </div>

  // Таблица платежей
  const showIns = insProp > 0 || insLife > 0;
  const showEarly = doEarly;

  const theadHtml = `

      <th style="padding:7px 8px;text-align:center;font-weight:600;white-space:nowrap;">№</th>

      <th style="padding:7px 8px;text-align:right;font-weight:600;white-space:nowrap;">Проценты</th>
      ${showEarly ? '<th style="padding:7px 8px;text-align:right;font-weight:600;white-space:nowrap;">Досрочно</th>' : ''}

    </tr>
  `;

  const tbodyHtml = rows.map(r => `
    <tr style="${r.early > 0 ? 'background:#1e2a1e;' : ''}">

  const tfootHtml = `
    <tr style="background:#313244;font-weight:700;">

      <td style="padding:8px;text-align:right;white-space:nowrap;">${formatMoney(principal)}</td>
      <td style="padding:8px;text-align:right;white-space:nowrap;color:#f38ba8;">${formatMoney(totalInterestPaid)}</td>
      ${showEarly ? `<td style="padding:8px;text-align:right;white-space:nowrap;color:#a6e3a1;">${formatMoney(totalEarlyPaid)}</td>` : ''}
      ${showIns   ? `<td style="padding:8px;text-align:right;white-space:nowrap;color:#fab387;">${formatMoney(totalInsPaid)}</td>` : ''}
      <td style="padding:8px;text-align:right;white-space:nowrap;color:#a6e3a1;">0,00 ₽</td>
    </tr>
  `;

  resultEl.innerHTML = `
    <div style="display:flex;flex-direction:column;gap:14px;margin-top:8px;">

          <thead>${theadHtml}</thead>
          <tbody>${tbodyHtml}</tbody>
          <tfoot>${tfootHtml}</tfoot>

      </div>
    </div>
  `;
});
